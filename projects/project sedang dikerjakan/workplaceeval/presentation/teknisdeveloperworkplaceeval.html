User Model Eloquent

class User extends Authenticatable { // 5 User Types protected $fillable = ['name', 'email', 'user_type', 'program_id']; // Relationships public function program() { return $this->belongsTo(Program::class); } public function placements() { return $this->hasMany(InternshipPlacement::class, 'student_id'); } public function assessments() { return $this->hasMany(Assessment::class, 'assessor_id'); } // Scopes public function scopeAdmin($q) { return $q->where('user_type', 'admin'); } public function scopeSupervisors($q) { return $q->whereIn('user_type', ['dosen', 'pembimbing_institusi', 'pembimbing_klinik']); } // Helpers public function isAdmin(): bool { return $this->user_type === 'admin'; } public function isSupervisor(): bool { return in_array($this->user_type, ['dosen', 'pembimbing_institusi', 'pembimbing_klinik']); } }

class Assessment extends Model { // Status: draft → submitted → approved/rejected public function calculateTotalScore(): float { match($this->template->calculation_method) { 'average' => $this->scores->avg('score_value'), 'weighted' => $this->calculateWeighted(), 'sum' => $this->scores->sum('score_value'), }; } public function determineFinalGrade(): string { $pct = ($this->total_score / $this->template->max_score) * 100; return match(true) { $pct >= 85 => 'A', $pct >= 75 => 'B', $pct >= 65 => 'C', $pct >= 55 => 'D', default => 'E', }; } public function submit(): bool { // draft → submitted } public function approve(User $admin): bool { // submitted → approved } public function reject(User $admin, string $reason): bool { // submitted → rejected } }

Assessment Policy Authorization

class AssessmentPolicy { public function view(User $user, Assessment $assessment): bool { if ($user->isAdmin()) return true; if ($user->isSupervisor() && $assessment->assessor_id === $user->id) return true; if ($user->isMahasiswa()) { return $assessment->placement->student_id === $user->id && $assessment->status === 'approved'; } return false; } public function update(User $user, Assessment $assessment): bool { return $user->id === $assessment->assessor_id && $assessment->status === 'draft'; } public function approve(User $user, Assessment $assessment): bool { return $user->isAdmin() && $assessment->status === 'submitted'; } }

Score Calculation Algorithms

📊 Average Method

$total = $scores->avg('score_value');

Simple average of all component scores. Best for equal-weight components.

⚖️ Weighted Method

$total = Σ(score × catWeight × compWeight) / Σweights;

Considers category and component weights. Best for differentiated importance.

Grade Determination

$percentage = ($totalScore / $maxScore) * 100; $grade = match(true) { $percentage >= 85 => 'A', // Excellent $percentage >= 75 => 'B', // Good $percentage >= 65 => 'C', // Satisfactory $percentage >= 55 => 'D', // Needs Improvement default => 'E', // Unsatisfactory }; $isPassed = $totalScore >= $template->passing_score;
