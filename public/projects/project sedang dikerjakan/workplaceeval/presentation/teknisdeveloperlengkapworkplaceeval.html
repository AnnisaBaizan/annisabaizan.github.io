📦 Assessment - Business Logic Methods

// ========== CALCULATION METHODS ========== public function calculateTotalScore(): float { $method = $this->template->calculation_method; return match($method) { 'average' => $this->calculateAverage(), 'weighted' => $this->calculateWeighted(), 'sum' => $this->calculateSum(), default => $this->calculateAverage(), }; } private function calculateAverage(): float { return $this->scores->avg('score_value') ?? 0; } private function calculateWeighted(): float { $totalWeightedScore = 0; $totalWeight = 0; foreach ($this->template->categories as $category) { $categoryWeight = $category->weight / 100; foreach ($category->components as $component) { $score = $this->scores->where('component_id', $component->id)->first(); if ($score) { $componentWeight = $component->weight / 100; $normalizedScore = ($score->score_value / $component->max_score) * 100; $totalWeightedScore += $normalizedScore * $categoryWeight * $componentWeight; $totalWeight += $categoryWeight * $componentWeight; } } } return $totalWeight > 0 ? $totalWeightedScore / $totalWeight : 0; } private function calculateSum(): float { return $this->scores->sum('score_value') ?? 0; }

📦 Assessment - Grading & Status Methods

// ========== GRADING ========== public function determineFinalGrade(): string { $percentage = ($this->total_score / $this->template->max_score) * 100; return match(true) { $percentage >= 85 => 'A', $percentage >= 75 => 'B', $percentage >= 65 => 'C', $percentage >= 55 => 'D', default => 'E', }; } public function isPassed(): bool { return $this->total_score >= $this->template->passing_score; } // ========== STATUS TRANSITIONS ========== public function submit(): bool { if ($this->status !== 'draft') return false; $this->total_score = $this->calculateTotalScore(); $this->final_grade = $this->determineFinalGrade(); $this->is_passed = $this->isPassed(); $this->status = 'submitted'; return $this->save(); } public function approve(User $admin): bool { if ($this->status !== 'submitted') return false; $this->status = 'approved'; $this->approved_by = $admin->id; $this->approved_at = now(); return $this->save(); } public function reject(User $admin, string $reason): bool { if ($this->status !== 'submitted') return false; $this->status = 'rejected'; $this->rejection_reason = $reason; return $this->save(); } }

🎮 Assessment Submit Calculation

public function submit($id) { $assessment = Assessment::with(['template', 'scores.component'])->findOrFail($id); $this->authorize('update', $assessment); // Validate all required components are filled $requiredComponents = $assessment->template->categories ->flatMap(fn($c) => $c->components->where('is_required', true)); $filledComponents = $assessment->scores->pluck('component_id'); $missingComponents = $requiredComponents ->filter(fn($c) => !$filledComponents->contains($c->id)); if ($missingComponents->isNotEmpty()) { return back()->withErrors([ 'scores' => 'Semua komponen wajib harus diisi sebelum submit' ]); } // Calculate and submit if ($assessment->submit()) { return redirect() ->route('assessor.assessments.show', $assessment) ->with('success', 'Penilaian berhasil disubmit. Total: ' . $assessment->total_score); } return back()->withErrors(['submit' => 'Gagal submit penilaian']); } }
